# 🧱 EC2 初始化

> ✅ **root 用户执行**

```
#!/usr/bin/env bash
set -e

echo "=============================="
echo " EC2 Base Environment Init "
echo "=============================="

### 0️⃣ 基础工具
echo ">>> Installing base tools..."
dnf update -y
dnf install -y --allowerasing \
  curl \
  wget \
  git \
  unzip \
  tar \
  yum-utils

# --------------------------------------------------
# 1️⃣ Java 21 LTS (Amazon Corretto)
# --------------------------------------------------
echo ">>> Installing Amazon Corretto 21..."
dnf install -y java-21-amazon-corretto-devel

java -version

# --------------------------------------------------
# 2️⃣ Node.js 20.19.0 + npm
# --------------------------------------------------
echo ">>> Installing Node.js 20.x..."
curl -fsSL https://rpm.nodesource.com/setup_20.x | bash -
dnf install -y nodejs

node -v
npm -v

# --------------------------------------------------
# 3️⃣ PM2
# --------------------------------------------------
echo ">>> Installing PM2..."
npm install -g pm2@6.0.5
pm2 -v

# --------------------------------------------------
# Docker (Amazon Linux 2023 官方)
# --------------------------------------------------
echo ">>> Installing Docker (Amazon Linux 2023)..."

dnf install -y docker

systemctl enable docker
systemctl start docker

docker version

# --------------------------------------------------
# 7️⃣ Docker 非 root（可选）
# --------------------------------------------------
if id ec2-user &>/dev/null; then
  usermod -aG docker ec2-user
fi

echo "=============================="
echo " EC2 Init Finished ✅"
echo " Re-login required for docker group"
echo "=============================="


# --------------------------------------------------
# 8️⃣ Nginx 1.26.x (Official)
# --------------------------------------------------
echo ">>> Installing Nginx..."

cat >/etc/yum.repos.d/nginx.repo <<'EOF'
[nginx-stable]
name=nginx stable repo
baseurl=https://nginx.org/packages/mainline/amzn/2023/$basearch/
gpgcheck=1
enabled=1
gpgkey=https://nginx.org/keys/nginx_signing.key
module_hotfixes=true
EOF

dnf install -y nginx

systemctl enable nginx
systemctl start nginx

nginx -v
```
