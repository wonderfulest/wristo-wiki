```
git clone git@gitee.com:muyids/tools.git

mkdir -p /data/wristo/watchface-apps && cd /data/wristo/watchface-apps
git clone git@gitee.com:muyids/SuperAlpha.git
git clone git@gitee.com:muyids/SuperBarrel.git

cd /data/wristo/
git clone git@gitee.com:muyids/wristo-resources.git
```
