## product_stats（当前实时快照）

语义：从“最近一次归档点”到现在的增量

CREATE TABLE `product_stats` (
  `product_id` int unsigned NOT NULL COMMENT 'products.id',
  `download` int unsigned NOT NULL DEFAULT 0 COMMENT '当前周期下载量',
  `purchase` int unsigned NOT NULL DEFAULT 0 COMMENT '当前周期购买量',
  `last_event_at` datetime(6) DEFAULT NULL COMMENT '最近一次统计事件时间',
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
    ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品统计实时快照（增量）';

## product_stat_history（月度事实表）

语义：每月一个产品一行，永不修改（或仅幂等更新）

CREATE TABLE `product_stat_history` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `product_id` int unsigned NOT NULL COMMENT 'products.id',
  `stat_month` char(7) NOT NULL COMMENT '统计月份 YYYY-MM',
  `download` int unsigned NOT NULL DEFAULT 0 COMMENT '当月下载量',
  `purchase` int unsigned NOT NULL DEFAULT 0 COMMENT '当月购买量',
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_product_month` (`product_id`, `stat_month`),
  KEY `idx_month` (`stat_month`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品统计历史（月度事实表）';

## product_stat_totals（历史累计锚点）
语义：已经归档进 history 的“历史总数”

CREATE TABLE `product_stat_totals` (
  `product_id` int unsigned NOT NULL COMMENT 'products.id',
  `total_download` bigint unsigned NOT NULL DEFAULT 0 COMMENT '历史累计下载量',
  `total_purchase` bigint unsigned NOT NULL DEFAULT 0 COMMENT '历史累计购买量',
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
    ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='产品统计累计总数（历史锚点）';
